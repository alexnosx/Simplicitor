# Simplicitor 2.1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship 2.1.0 with PDF output from a prompt, a per-user MSI installer replacing `Simplicitor-setup.exe`, and a redesigned UI with exactly two modes, Create new and Edit document.

**Architecture:** Part A adds a `PdfGenerator` that renders the existing Word response structure through Qt's bundled `QTextDocument` and `QPdfWriter`; no new dependency and no Office. Part B builds an MSI from the Nuitka standalone payload with a pinned, repo-local WiX tool; `build.py` generates the per-folder component list so per-user install and uninstall are clean, and the hosted runner qualifies install, repair, upgrade and uninstall. Part C redesigns the mockup first (C1) and, only after Alex approves it, rebuilds the Qt screens (C2) around one step-based Create screen with a complete, specified interface.

**Tech Stack:** Python 3.12 locally (`.venv`), Python 3.11 on GitHub Actions; PySide6 6.11.2 (QtGui `QTextDocument`, `QPdfWriter`); pypdf (already a dependency); Nuitka 4.2.2 standalone; WiX Toolset 5.0.2 as a .NET local tool; GitHub Actions `windows-latest`; pytest and pytest-qt.

**Status:** Revision 2, 2026-10-09. Drafted by Claude from Alex's decisions; revised after the [Codex review](../../reviews/2026-10-09-release-2.1.0-codex.md). Nothing in this plan authorizes a tag, release, or publication. Alex authorizes execution part by part.

```python
# simplicitor/app/generators/pdf_generator.py
"""Render a parsed Word response to PDF with Qt's bundled PDF writer."""
import html
import logging
import os
import tempfile
from pathlib import Path

from pypdf import PdfReader
from pypdf.errors import PyPdfError
from PySide6.QtCore import QMarginsF
from PySide6.QtGui import QPageLayout, QPageSize, QPdfWriter, QTextDocument

logger = logging.getLogger(__name__)


def _print_document(document: QTextDocument, writer: QPdfWriter) -> None:
    """Separate seam so tests can simulate a write failure."""
    document.print_(writer)


def _validate_pdf(path: Path) -> None:
    """Reject empty or truncated output before it replaces anything."""
    try:
        if not PdfReader(str(path)).pages:
            raise OSError("PDF output has no pages.")
    except PyPdfError as exc:
        raise OSError("PDF output is incomplete.") from exc


def _split_row(line: str) -> list[str]:
    if "|" in line:
        return [cell.strip() for cell in line.strip().strip("|").split("|")]
    return [cell.strip() for cell in line.split("\t")]


def _escape(text: str) -> str:
    return html.escape(text).replace("\n", "<br>")


class PdfGenerator:
    """Generates .pdf files from the same parsed structure as WordGenerator."""

    def generate(self, parsed: dict, output_path: Path) -> Path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        document = QTextDocument()
        document.setHtml(self._html(parsed))
        descriptor, name = tempfile.mkstemp(prefix=".simplicitor-", suffix=".pdf",
                                            dir=output_path.parent)
        os.close(descriptor)
        temporary = Path(name)
        writer = None
        try:
            writer = QPdfWriter(str(temporary))
            writer.setPageSize(QPageSize(QPageSize.PageSizeId.A4))
            writer.setPageMargins(QMarginsF(20, 20, 20, 20), QPageLayout.Unit.Millimeter)
            _print_document(document, writer)
            writer = None  # Releases the writer and closes the file before validation.
            _validate_pdf(temporary)
            os.replace(temporary, output_path)
        except OSError as exc:
            # Metadata only: the filename can contain confidential prompt words.
            logger.error("PDF generation failed: %s", type(exc).__name__)
            raise
        finally:
            writer = None
            temporary.unlink(missing_ok=True)
        return output_path

    def _html(self, parsed: dict) -> str:
        parts = [f"<h1>{_escape(parsed.get('title', ''))}</h1>"]
        for section in parsed.get("sections", []):
            heading = section.get("heading", "")
            if heading:
                parts.append(f"<h2>{_escape(heading)}</h2>")
            content = section.get("content", "")
            kind = section.get("type", "text")
            if kind == "list":
                items = [ln.strip() for ln in content.replace("\n\n", "\n").split("\n") if ln.strip()]
                parts.append("<ul>" + "".join(f"<li>{_escape(i)}</li>" for i in items) + "</ul>")
            elif kind == "table":
                rows = [_split_row(ln) for ln in content.split("\n") if ln.strip()]
                if rows:
                    head = "".join(f"<th>{_escape(c)}</th>" for c in rows[0])
                    body = "".join("<tr>" + "".join(f"<td>{_escape(c)}</td>" for c in r) + "</tr>"
                                   for r in rows[1:])
                    parts.append("<table border='1' cellspacing='0' cellpadding='4'>"
                                 f"<tr>{head}</tr>{body}</table>")
            else:
                for chunk in content.split("\n\n"):
                    if chunk.strip():
                        parts.append(f"<p>{_escape(chunk.strip())}</p>")
        return "".join(parts)
```

```python
from xml.sax.saxutils import quoteattr


def _wix_id(prefix: str, relative: str) -> str:
    return prefix + hashlib.sha256(relative.encode("utf-8")).hexdigest()[:40]


def payload_fragment(payload: Path) -> str:
    """WiX fragment: one per-user component per payload folder (HKCU key path,
    the folder's files, and its RemoveFolder) under INSTALLFOLDER."""
    folders = [payload] + sorted((p for p in payload.rglob("*") if p.is_dir()),
                                 key=lambda p: p.relative_to(payload).as_posix())
    rel = {f: (f.relative_to(payload).as_posix() or ".") for f in folders}
    dir_id = {f: ("INSTALLFOLDER" if f == payload else _wix_id("d", rel[f])) for f in folders}
    lines = ['<Wix xmlns="http://wixtoolset.org/schemas/v4/wxs">', "  <Fragment>",
             '    <DirectoryRef Id="INSTALLFOLDER">']

    def nest(folder: Path, depth: int) -> None:
        for child in sorted(p for p in folder.iterdir() if p.is_dir()):
            pad = "  " * depth
            lines.append(f'{pad}<Directory Id="{dir_id[child]}" Name={quoteattr(child.name)}>')
            nest(child, depth + 1)
            lines.append(f"{pad}</Directory>")

    nest(payload, 3)
    lines += ["    </DirectoryRef>", '    <ComponentGroup Id="Payload">']
    for folder in folders:
        lines.append(f'      <Component Id="{_wix_id("c", rel[folder])}" Directory="{dir_id[folder]}">')
        lines.append('        <RegistryValue Root="HKCU" Key="Software\\Simplicitor\\Installer\\Payload" '
                     f'Name={quoteattr(rel[folder])} Type="integer" Value="1" KeyPath="yes" />')
        lines.append(f'        <RemoveFolder Id="{_wix_id("r", rel[folder])}" On="uninstall" />')
        for file in sorted(p for p in folder.iterdir() if p.is_file()):
            relative = file.relative_to(payload).as_posix()
            source = "!(bindpath.payload)\\" + relative.replace("/", "\\")
            lines.append(f'        <File Id="{_wix_id("f", relative)}" Source={quoteattr(source)} />')
        lines.append("      </Component>")
    lines += ["    </ComponentGroup>", "  </Fragment>", "</Wix>"]
    return "\n".join(lines) + "\n"
```

```python
# simplicitor/app/widgets/step_indicator.py
"""Step names with the current step highlighted. Rebuilds only when the route changes."""
from PySide6.QtWidgets import QHBoxLayout, QLabel, QWidget

from app.config.defaults import PRIMARY_ACCENT_COLOR


class StepIndicator(QWidget):
    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self._layout = QHBoxLayout(self)
        self._layout.setContentsMargins(0, 0, 0, 0)
        self._layout.setSpacing(16)
        self._layout.addStretch()  # The only stretch; labels are inserted before it.
        self._labels: list[QLabel] = []
        self._names: list[str] = []
        self._current = 0
        self.setStyleSheet(
            f"QLabel[current='true'] {{ color: {PRIMARY_ACCENT_COLOR}; font-weight: 600; }}"
            "QLabel[current='false'] { color: #626B78; }")

    def set_steps(self, names: list[str]) -> None:
        if names == self._names:
            return
        for label in self._labels:
            self._layout.removeWidget(label)
            label.deleteLater()
        self._labels, self._names = [], list(names)
        for number, name in enumerate(names, 1):
            label = QLabel(name)
            label.setAccessibleName(f"Step {number}: {name}")
            self._layout.insertWidget(len(self._labels), label)
            self._labels.append(label)
        self.set_current(0)

    def current(self) -> int:
        return self._current

    def set_current(self, index: int) -> None:
        self._current = index
        for position, label in enumerate(self._labels):
            label.setProperty("current", position == index)
            label.style().unpolish(label)
            label.style().polish(label)
```

```python
"""Plain-language text for source locations, review reasons, and coverage."""
import re

_ANCHOR = re.compile(
    r"#(?:(header|footer):\d+:)?(?:p:(\d+)|t:(\d+):r:(\d+):c:(\d+)|page:(\d+))$")

_AI_INCOMPLETE = "The AI request did not complete. Check this value."
_ISSUES = {
    "sectioned_source": "Long document: check every value.",
    "missing_value": "Not found in the document.",
    "value_not_verbatim": "The value does not match the document text exactly.",
    "quote_not_in_source": "The quoted text was not found in the document.",
    "unknown_anchor": "The quoted text was not found in the document.",
    "invalid_value_type": "The AI returned this value in an unexpected form.",
    "conversion_failed": "Could not read this as a number or date.",
    "leading_zero": "Starts with 0, so it was kept as text to keep every digit.",
    "excel_precision": "Too many digits for an Excel number, so it was kept as text.",
    "conflict": "The document gives different values in different places.",
    "unverified_proposal": "Could not confirm this value from the document.",
    "context_truncated": _AI_INCOMPLETE,
    "output_limit": _AI_INCOMPLETE,
    "request_failed": _AI_INCOMPLETE,
    "request_timeout": _AI_INCOMPLETE,
    "invalid_response_schema": _AI_INCOMPLETE,
    "invalid_field_schema": _AI_INCOMPLETE,
}
_COVERAGE = {
    "unreadable_page": "this page has too little text to read.",
    "unsupported_structure": "some parts of this file (such as text boxes, tracked changes "
                             "or footnotes) were not read.",
    "oversized_unit": "one part of this file was too large to read.",
    "context_budget": "this file could not be read with the current settings.",
    "context_truncated": "part of this file was cut off and not read.",
}


def describe_anchor(anchor: str) -> str:
    match = _ANCHOR.search(anchor or "")
    if not match:
        return ""
    area, paragraph, table, row, column, page = match.groups()
    if page:
        where = f"page {page}"
    elif paragraph is not None:
        where = f"paragraph {int(paragraph) + 1}"
    else:
        where = f"table {int(table) + 1}, row {int(row) + 1}, column {int(column) + 1}"
    return f"{where} of the {area}" if area else where


def describe_issues(issues: str) -> str:
    codes = [code.strip() for code in (issues or "").split(";") if code.strip()]
    return " ".join(dict.fromkeys(_ISSUES.get(code, "Check this value.") for code in codes))


def describe_coverage(file_label: str, anchor: str, code: str) -> str:
    code = code.split(":", 1)[0].strip()
    reason = _COVERAGE.get(code, "part of this file was not read.")
    where = describe_anchor(anchor)
    return f"{file_label}, {where}: {reason}" if where else f"{file_label}: {reason}"
```

```python
            model.item(index).setEnabled(not has_sources or self._type_combo.itemText(index) == EXCEL)

            self._primary_btn.setEnabled(ready)

        if self._paths:
            self.propose_requested.emit(self.snapshot(include_columns=False))
        else:
            self.generate_requested.emit(self._type_combo.currentText(), self._save_dir,
                                         self._prompt_edit.toPlainText().strip())
```
