# Simplicitor 2.1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship 2.1.0 with PDF output from a prompt, a per-user MSI installer replacing `Simplicitor-setup.exe`, and a redesigned UI with exactly two modes, Create new and Edit document.

**Architecture:** Three independent parts. Part A adds a `PdfGenerator` that renders the existing Word response structure through Qt's bundled `QTextDocument` and `QPdfWriter`; no new dependency and no Office. Part B builds an MSI with a pinned, repo-local WiX tool from the same Nuitka standalone payload, and moves the hosted-runner qualification to `msiexec`. Part C redesigns the UI mockup first (C1), then, only after Alex approves it, rebuilds the Qt screens (C2) by moving the existing extraction widgets into a step-based Create screen.

**Tech Stack:** Python 3.12, PySide6 6.11.2 (QtGui `QTextDocument`, `QPdfWriter`), Nuitka 4.2.2 standalone, WiX Toolset as a .NET local tool, GitHub Actions `windows-latest`, pytest and pytest-qt.

**Author and status:** Drafted by Claude for Codex review on 2026-10-09, from Alex's decisions the same day. Codex: review this plan and report disagreements before executing. Nothing in this plan authorizes a tag, release, or publication.

```python
# simplicitor/app/generators/pdf_generator.py
"""Render a parsed Word response to PDF with Qt's bundled PDF writer."""
import html
import logging
import os
import tempfile
from pathlib import Path

from PySide6.QtCore import QMarginsF
from PySide6.QtGui import QPageLayout, QPageSize, QPdfWriter, QTextDocument

logger = logging.getLogger(__name__)


def _print_document(document: QTextDocument, writer: QPdfWriter) -> None:
    """Separate seam so tests can simulate a write failure."""
    document.print_(writer)


def _split_row(line: str) -> list[str]:
    if "|" in line:
        return [cell.strip() for cell in line.strip().strip("|").split("|")]
    return [cell.strip() for cell in line.split("\t")]


class PdfGenerator:
    """Generates .pdf files from the same parsed structure as WordGenerator."""

    def generate(self, parsed: dict, output_path: Path) -> Path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        document = QTextDocument()
        document.setHtml(self._html(parsed))
        descriptor, name = tempfile.mkstemp(prefix=".simplicitor-", suffix=".pdf",
                                            dir=output_path.parent)
        os.close(descriptor)
        temporary = Path(name)
        try:
            writer = QPdfWriter(str(temporary))
            writer.setPageSize(QPageSize(QPageSize.PageSizeId.A4))
            writer.setPageMargins(QMarginsF(20, 20, 20, 20), QPageLayout.Unit.Millimeter)
            _print_document(document, writer)
            del writer  # Closes the PDF file before it is checked and moved.
            if not temporary.read_bytes().startswith(b"%PDF"):
                raise OSError("PDF writer produced no output.")
            os.replace(temporary, output_path)
        except OSError as exc:
            logger.error("Failed to write PDF %s: %s", output_path, type(exc).__name__)
            raise
        finally:
            temporary.unlink(missing_ok=True)
        return output_path

    def _html(self, parsed: dict) -> str:
        parts = [f"<h1>{html.escape(parsed.get('title', ''))}</h1>"]
        for section in parsed.get("sections", []):
            heading = section.get("heading", "")
            if heading:
                parts.append(f"<h2>{html.escape(heading)}</h2>")
            content = section.get("content", "")
            kind = section.get("type", "text")
            if kind == "list":
                items = [ln.strip() for ln in content.replace("\n\n", "\n").split("\n") if ln.strip()]
                parts.append("<ul>" + "".join(f"<li>{html.escape(i)}</li>" for i in items) + "</ul>")
            elif kind == "table":
                rows = [_split_row(ln) for ln in content.split("\n") if ln.strip()]
                if rows:
                    head = "".join(f"<th>{html.escape(c)}</th>" for c in rows[0])
                    body = "".join("<tr>" + "".join(f"<td>{html.escape(c)}</td>" for c in r) + "</tr>"
                                   for r in rows[1:])
                    parts.append("<table border='1' cellspacing='0' cellpadding='4'>"
                                 f"<tr>{head}</tr>{body}</table>")
            else:
                for chunk in content.split("\n\n"):
                    if chunk.strip():
                        parts.append(f"<p>{html.escape(chunk.strip())}</p>")
        return "".join(parts)
```

```python
    """Compile the per-user MSI from the standalone payload with the pinned WiX tool."""
    restore = subprocess.run(["dotnet", "tool", "restore"], cwd=ROOT)
```

```python
"""Numbered steps; the current step uses the accent color."""
from PySide6.QtWidgets import QHBoxLayout, QLabel, QWidget

from app.config.defaults import PRIMARY_ACCENT_COLOR


class StepIndicator(QWidget):
    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self._layout = QHBoxLayout(self)
        self._layout.setContentsMargins(0, 0, 0, 0)
        self._layout.setSpacing(16)
        self._labels: list[QLabel] = []
        self.setStyleSheet(
            f"QLabel[current='true'] {{ color: {PRIMARY_ACCENT_COLOR}; font-weight: 600; }}"
            "QLabel[current='false'] { color: #626B78; }")

    def set_steps(self, names: list[str]) -> None:
        for label in self._labels:
            self._layout.removeWidget(label)
            label.deleteLater()
        self._labels = []
        for number, name in enumerate(names, 1):
            label = QLabel(name)
            label.setAccessibleName(f"Step {number}: {name}")
            self._labels.append(label)
            self._layout.addWidget(label)
        self._layout.addStretch()
        self.set_current(0)

    def set_current(self, index: int) -> None:
        for position, label in enumerate(self._labels):
            label.setProperty("current", position == index)
            label.style().unpolish(label)
            label.style().polish(label)
```

```python
"""Plain-language text for source locations and review reasons."""
import re

_ANCHOR = re.compile(
    r"#(?:(header|footer):\d+:)?(?:p:(\d+)|t:(\d+):r:(\d+):c:(\d+)|page:(\d+))$")

_ISSUES = {
    "sectioned_source": "Long document: check every value.",
    "missing_value": "Not found in the document.",
    "value_not_verbatim": "The value does not match the document text exactly.",
    "quote_not_in_source": "The quoted text was not found in the document.",
    "unknown_anchor": "The quoted text was not found in the document.",
    "conversion_failed": "Could not read this as a number or date.",
    "leading_zero": "Starts with 0, so it was kept as text to keep every digit.",
    "excel_precision": "Too many digits for an Excel number, so it was kept as text.",
    "conflict": "The document gives different values in different places.",
    "unverified_proposal": "Could not confirm this value from the document.",
    "context_truncated": "The AI request did not complete. Check this value.",
    "output_limit": "The AI request did not complete. Check this value.",
    "request_failed": "The AI request did not complete. Check this value.",
    "request_timeout": "The AI request did not complete. Check this value.",
    "invalid_response_schema": "The AI request did not complete. Check this value.",
    "invalid_field_schema": "The AI request did not complete. Check this value.",
}


def describe_anchor(anchor: str) -> str:
    match = _ANCHOR.search(anchor or "")
    if not match:
        return ""
    area, paragraph, table, row, column, page = match.groups()
    if page:
        where = f"page {page}"
    elif paragraph is not None:
        where = f"paragraph {int(paragraph) + 1}"
    else:
        where = f"table {int(table) + 1}, row {int(row) + 1}, column {int(column) + 1}"
    return f"{where} of the {area}" if area else where


def describe_issues(issues: str) -> str:
    codes = [code.strip() for code in (issues or "").split(";") if code.strip()]
    texts = list(dict.fromkeys(_ISSUES.get(code, "Check this value.") for code in codes))
    return " ".join(texts)
```

```python
        super().__init__(parent)
        self._settings = settings

        self._sources_note.setWordWrap(True)
        column.addWidget(self._sources_note)
        actions = QHBoxLayout()
        actions.addStretch()
        self._primary_btn = QPushButton()
        self._primary_btn.clicked.connect(self._primary_clicked)
        actions.addWidget(self._primary_btn)
        column.addLayout(actions)
        return page

    def set_paths(self, paths: tuple[Path, ...]) -> None:
        self._paths = tuple(paths)
        self._source_list.clear()
        self._source_list.addItems([p.name for p in self._paths])
        self.setup_changed.emit()
        if self._paths:
            self.inspect_requested.emit(self.snapshot(include_columns=False))
        self._refresh()

    def set_model(self, model: str) -> None:
        self._model = model
        self._refresh()

    def set_ollama_connected(self, connected: bool) -> None:
        self._connected = connected
        self._refresh()

    def _type_changed(self, label: str) -> None:
        self._prompt_edit.setPlaceholderText(PROMPT_PLACEHOLDERS.get(label, ""))
        self.file_type_changed.emit(label)

    def _refresh(self) -> None:
        has_sources = bool(self._paths)
        model = self._type_combo.model()
        for index in range(self._type_combo.count()):
            model.item(index).setEnabled(not has_sources or self._type_combo.itemText(index) == EXCEL)
        if has_sources and self._type_combo.currentText() != EXCEL:
            self._type_combo.setCurrentText(EXCEL)
        self._sources_note.setVisible(has_sources)
        self._primary_btn.setText("Next: choose columns" if has_sources else "Create file")

                              else ["Describe", "Save"])

    def _primary_clicked(self) -> None:
        prompt = self._prompt_edit.toPlainText().strip()
        if self._paths:
            self.propose_requested.emit(self.snapshot(include_columns=False))
        else:
            self.generate_requested.emit(self._type_combo.currentText(),
                                         self._settings.generated_dir, prompt)
```
